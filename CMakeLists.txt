cmake_minimum_required(VERSION 3.22)

project(KyotoSpxrit VERSION 0.2.2 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CMAKE_OSX_DEPLOYMENT_TARGET "10.13" CACHE STRING "Minimum macOS version")

# ============================================================
# JUCE
# ============================================================

include(FetchContent)

FetchContent_Declare(
    JUCE
    GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG 8.0.6
    GIT_SHALLOW TRUE
)

FetchContent_MakeAvailable(JUCE)

# ============================================================
# KyotoSpxrit source files
# ============================================================

set(KYOTO_SOURCES
    Source/PluginProcessor.cpp
    Source/PluginEditor.cpp
    Source/InstrumentEngine.cpp
    Source/DreamAPI.cpp
    Source/UpdateService.cpp
    Source/ModuleRegistry.cpp
    Source/DspEngine.cpp
    Source/ThemeManager.cpp
    Source/PresetCodec.cpp
    Source/BuilderModel.h
)

# ============================================================
# Native BinaryData replacement
#
# JUCE's juceaide BinaryData generator was crashing on both
# Windows and macOS during the GitHub Actions build.
#
# We keep the exact BinaryData.h interface expected by the
# plugin, but generate it directly with CMake instead.
# ============================================================

set(KYOTO_RESOURCES_DIR
    "${CMAKE_CURRENT_SOURCE_DIR}/Resources"
)

set(KYOTO_THEMES_JSON
    "${KYOTO_RESOURCES_DIR}/themes/themes.json"
)

if(NOT EXISTS "${KYOTO_THEMES_JSON}")
    message(FATAL_ERROR
        "KyotoSpxrit resource file was not found: ${KYOTO_THEMES_JSON}"
    )
endif()

set(KYOTO_BINARYDATA_DIR
    "${CMAKE_CURRENT_BINARY_DIR}/KyotoBinaryData"
)

file(MAKE_DIRECTORY "${KYOTO_BINARYDATA_DIR}")

set(KYOTO_BINARYDATA_CPP
    "${KYOTO_BINARYDATA_DIR}/BinaryData.cpp"
)

set(KYOTO_BINARYDATA_H
    "${KYOTO_BINARYDATA_DIR}/BinaryData.h"
)

# Read the JSON as raw hexadecimal bytes.
file(
    READ
    "${KYOTO_THEMES_JSON}"
    KYOTO_THEMES_HEX
    HEX
)

# Convert every pair of hexadecimal digits into a C++ byte.
string(
    REGEX REPLACE
    "([0-9A-Fa-f][0-9A-Fa-f])"
    "0x\\1,"
    KYOTO_THEMES_BYTES
    "${KYOTO_THEMES_HEX}"
)

# Calculate the exact byte count.
string(
    LENGTH
    "${KYOTO_THEMES_HEX}"
    KYOTO_THEMES_HEX_LENGTH
)

math(
    EXPR
    KYOTO_THEMES_SIZE
    "${KYOTO_THEMES_HEX_LENGTH} / 2"
)

# Generate BinaryData.h.
file(
    WRITE
    "${KYOTO_BINARYDATA_H}"
    "/*
    This file is automatically generated by CMake.
    Do not edit manually.
*/

#pragma once

namespace BinaryData
{
    extern const unsigned char themes_json[];
    extern const int themes_jsonSize;
}
"
)

# Generate BinaryData.cpp.
file(
    WRITE
    "${KYOTO_BINARYDATA_CPP}"
    "/*
    This file is automatically generated by CMake.
    Do not edit manually.
*/

#include \"BinaryData.h\"

namespace BinaryData
{
    const unsigned char themes_json[] =
    {
        ${KYOTO_THEMES_BYTES}
    };

    const int themes_jsonSize = ${KYOTO_THEMES_SIZE};
}
"
)

# Static library exposing the same BinaryData API expected by
# PluginProcessor.cpp.
add_library(
    BinaryData
    STATIC
    "${KYOTO_BINARYDATA_CPP}"
)

target_include_directories(
    BinaryData
    PUBLIC
    "${KYOTO_BINARYDATA_DIR}"
)

set_target_properties(
    BinaryData
    PROPERTIES
    POSITION_INDEPENDENT_CODE TRUE
)

target_compile_features(
    BinaryData
    PRIVATE
    cxx_std_17
)

# ============================================================
# Instrument plugin
# ============================================================

juce_add_plugin(
    KyotoSpxrit

    COMPANY_NAME "KyotoSpxrit"

    BUNDLE_ID "com.kyotospxrit.instrument"

    PLUGIN_MANUFACTURER_CODE Kyto
    PLUGIN_CODE KyD1

    FORMATS VST3

    PRODUCT_NAME "KyotoSpxrit"

    VST3_CATEGORIES Instrument Synth

    IS_SYNTH TRUE
    NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT FALSE

    EDITOR_WANTS_KEYBOARD_FOCUS TRUE

    COPY_PLUGIN_AFTER_BUILD FALSE
)

# Use apostrophe-free target/output names on every platform. This avoids
# the Windows MSBuild path parsing issue and the macOS/Xcode shared-code
# archive mismatch that occurred with the old product name.
set_target_properties(KyotoSpxrit PROPERTIES OUTPUT_NAME "KyotoSpxrit")

juce_generate_juce_header(KyotoSpxrit)

target_sources(
    KyotoSpxrit
    PRIVATE
    ${KYOTO_SOURCES}
)

target_link_libraries(
    KyotoSpxrit
    PRIVATE

    BinaryData

    juce::juce_audio_utils

    juce::juce_recommended_config_flags
    juce::juce_recommended_lto_flags
    juce::juce_recommended_warning_flags
)

target_compile_definitions(
    KyotoSpxrit
    PRIVATE

    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_DISPLAY_SPLASH_SCREEN=0
    JUCE_STRICT_REFCOUNTEDPOINTER=1
)

# ============================================================
# FX plugin
# ============================================================

juce_add_plugin(
    KyotoSpxritFX

    COMPANY_NAME "KyotoSpxrit"

    BUNDLE_ID "com.kyotospxrit.fx"

    PLUGIN_MANUFACTURER_CODE Kyto
    PLUGIN_CODE KyD2

    FORMATS VST3

    PRODUCT_NAME "KyotoSpxrit FX"

    VST3_CATEGORIES Fx

    IS_SYNTH FALSE
    NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT FALSE

    EDITOR_WANTS_KEYBOARD_FOCUS TRUE

    COPY_PLUGIN_AFTER_BUILD FALSE
)

# Same cross-platform output-name fix for the FX target.
set_target_properties(KyotoSpxritFX PROPERTIES OUTPUT_NAME "KyotoSpxritFX")

juce_generate_juce_header(KyotoSpxritFX)

target_sources(
    KyotoSpxritFX
    PRIVATE
    ${KYOTO_SOURCES}
)

target_link_libraries(
    KyotoSpxritFX
    PRIVATE

    BinaryData

    juce::juce_audio_utils

    juce::juce_recommended_config_flags
    juce::juce_recommended_lto_flags
    juce::juce_recommended_warning_flags
)

target_compile_definitions(
    KyotoSpxritFX
    PRIVATE

    KYOTO_IS_FX=1

    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_DISPLAY_SPLASH_SCREEN=0
    JUCE_STRICT_REFCOUNTEDPOINTER=1
)

# ============================================================
# Install both VST3 targets
# ============================================================

install(
    TARGETS
    KyotoSpxrit
    KyotoSpxritFX

    LIBRARY DESTINATION .
    RUNTIME DESTINATION .
    ARCHIVE DESTINATION .
)
